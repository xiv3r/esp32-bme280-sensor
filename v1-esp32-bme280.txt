void applyDefaults() {
  strncpy(wifiConfig.ssid, DEFAULT_AP_SSID, 32);
  wifiConfig.ssid[32] = 0;
  strncpy(wifiConfig.pass, DEFAULT_AP_PASS, 63);
  wifiConfig.pass[63] = 0;
  settings.tempMin = -10.0f;
  settings.tempMax = 40.0f;
  settings.humMin = 20.0f;
  settings.humMax = 80.0f;
  settings.presMin = 950.0f;
  settings.presMax = 1050.0f;
  settings.altRef = 1013.25f;
  settings.relayMode = 0;
  relayState = false;
}

bool settingsValid(const Settings &s) {
  if (!(s.tempMin < s.tempMax)) return false;
  if (!(s.humMin < s.humMax)) return false;
  if (!(s.presMin < s.presMax)) return false;
  if (s.humMin < 0.0f || s.humMax > 100.0f) return false;
  if (s.altRef < 300.0f || s.altRef > 1100.0f) return false;
  if (s.relayMode > 4) return false;
  if (s.tempMin < -100.0f || s.tempMax > 200.0f) return false;
  if (s.presMin < 300.0f || s.presMax > 1200.0f) return false;
  return true;
}

bool wifiValid() {
  size_t sl = strlen(wifiConfig.ssid);
  size_t pl = strlen(wifiConfig.pass);
  if (sl < 1 || sl > 32) return false;
  if (pl != 0 && (pl < 8 || pl > 63)) return false;
  for (size_t i = 0; i < sl; i++) {
    unsigned char c = (unsigned char)wifiConfig.ssid[i];
    if (c < 0x20 || c == 0x7f || c == '"' || c == '\\') return false;
  }
  for (size_t i = 0; i < pl; i++) {
    unsigned char c = (unsigned char)wifiConfig.pass[i];
    if (c < 0x20 || c == 0x7f || c == '"' || c == '\\') return false;
  }
  return true;
}

void nvsLoadWifi() {
  prefs.begin(NVS_NS_WIFI, true);
  uint32_t magic = prefs.getUInt(KEY_CONFIG_VALID, 0);
  if (magic != CONFIG_MAGIC) {
    prefs.end();
    strncpy(wifiConfig.ssid, DEFAULT_AP_SSID, 32);
    wifiConfig.ssid[32] = 0;
    strncpy(wifiConfig.pass, DEFAULT_AP_PASS, 63);
    wifiConfig.pass[63] = 0;
    return;
  }
  String s = prefs.getString(KEY_WIFI_SSID, DEFAULT_AP_SSID);
  String p = prefs.getString(KEY_WIFI_PASS, DEFAULT_AP_PASS);
  prefs.end();
  strncpy(wifiConfig.ssid, s.c_str(), 32);
  wifiConfig.ssid[32] = 0;
  strncpy(wifiConfig.pass, p.c_str(), 63);
  wifiConfig.pass[63] = 0;
  if (!wifiValid()) {
    strncpy(wifiConfig.ssid, DEFAULT_AP_SSID, 32);
    wifiConfig.ssid[32] = 0;
    strncpy(wifiConfig.pass, DEFAULT_AP_PASS, 63);
    wifiConfig.pass[63] = 0;
  }
}

void nvsSaveWifi() {
  prefs.begin(NVS_NS_WIFI, false);
  prefs.putString(KEY_WIFI_SSID, wifiConfig.ssid);
  prefs.putString(KEY_WIFI_PASS, wifiConfig.pass);
  prefs.putUInt(KEY_CONFIG_VALID, CONFIG_MAGIC);
  prefs.end();
}

void nvsLoadSettings() {
  prefs.begin(NVS_NS_SENSOR, true);
  uint32_t magic = prefs.getUInt(KEY_CONFIG_VALID, 0);
  if (magic != CONFIG_MAGIC) {
    prefs.end();
    settings.tempMin = -10.0f;
    settings.tempMax = 40.0f;
    settings.humMin = 20.0f;
    settings.humMax = 80.0f;
    settings.presMin = 950.0f;
    settings.presMax = 1050.0f;
    settings.altRef = 1013.25f;
    settings.relayMode = 0;
    return;
  }
  settings.tempMin = prefs.getFloat(KEY_TEMP_MIN, -10.0f);
  settings.tempMax = prefs.getFloat(KEY_TEMP_MAX, 40.0f);
  settings.humMin = prefs.getFloat(KEY_HUM_MIN, 20.0f);
  settings.humMax = prefs.getFloat(KEY_HUM_MAX, 80.0f);
  settings.presMin = prefs.getFloat(KEY_PRES_MIN, 950.0f);
  settings.presMax = prefs.getFloat(KEY_PRES_MAX, 1050.0f);
  settings.altRef = prefs.getFloat(KEY_ALT_REF, 1013.25f);
  settings.relayMode = prefs.getUChar(KEY_RELAY_MODE, 0);
  prefs.end();
  if (!settingsValid(settings)) {
    settings.tempMin = -10.0f;
    settings.tempMax = 40.0f;
    settings.humMin = 20.0f;
    settings.humMax = 80.0f;
    settings.presMin = 950.0f;
    settings.presMax = 1050.0f;
    settings.altRef = 1013.25f;
    settings.relayMode = 0;
  }
}

void nvsSaveSettings() {
  prefs.begin(NVS_NS_SENSOR, false);
  prefs.putFloat(KEY_TEMP_MIN, settings.tempMin);
  prefs.putFloat(KEY_TEMP_MAX, settings.tempMax);
  prefs.putFloat(KEY_HUM_MIN, settings.humMin);
  prefs.putFloat(KEY_HUM_MAX, settings.humMax);
  prefs.putFloat(KEY_PRES_MIN, settings.presMin);
  prefs.putFloat(KEY_PRES_MAX, settings.presMax);
  prefs.putFloat(KEY_ALT_REF, settings.altRef);
  prefs.putUChar(KEY_RELAY_MODE, settings.relayMode);
  prefs.putUInt(KEY_CONFIG_VALID, CONFIG_MAGIC);
  prefs.end();
}

void nvsLoadRelayState() {
  prefs.begin(NVS_NS_RELAY, true);
  uint32_t magic = prefs.getUInt(KEY_CONFIG_VALID, 0);
  if (magic != CONFIG_MAGIC) {
    prefs.end();
    relayState = false;
    return;
  }
  relayState = prefs.getBool(KEY_RELAY_STATE, false);
  prefs.end();
}

void nvsSaveRelayState() {
  prefs.begin(NVS_NS_RELAY, false);
  prefs.putBool(KEY_RELAY_STATE, relayState);
  prefs.putUInt(KEY_CONFIG_VALID, CONFIG_MAGIC);
  prefs.end();
}

void nvsFactoryReset() {
  prefs.begin(NVS_NS_WIFI, false); prefs.clear(); prefs.end();
  prefs.begin(NVS_NS_SENSOR, false); prefs.clear(); prefs.end();
  prefs.begin(NVS_NS_RELAY, false); prefs.clear(); prefs.end();
  applyDefaults();
  nvsSaveWifi();
  nvsSaveSettings();
  nvsSaveRelayState();
}

void setRelay(bool on) {
  if (relayState == on) return;
  relayState = on;
  digitalWrite(RELAY_PIN, on ? HIGH : LOW);
  relayPersistPending = true;
  relayPersistDeadline = millis() + RELAY_SAVE_DEBOUNCE_MS;
}

void evaluateRelay() {
  if (!sensorData.valid) return;
  bool tOut = (sensorData.temperature < settings.tempMin || sensorData.temperature > settings.tempMax);
  bool hOut = (sensorData.humidity < settings.humMin || sensorData.humidity > settings.humMax);
  bool pOut = (sensorData.pressure < settings.presMin || sensorData.pressure > settings.presMax);
  bool trigger = false;
  switch (settings.relayMode) {
    case 0: trigger = tOut || hOut || pOut; break;
    case 1: trigger = tOut; break;
    case 2: trigger = hOut; break;
    case 3: trigger = pOut; break;
    case 4: trigger = tOut && hOut && pOut; break;
    default: trigger = false; break;
  }
  setRelay(trigger);
}

void readSensor() {
  unsigned long now = millis();
  if ((unsigned long)(now - lastSensorRead) < SENSOR_READ_INTERVAL_MS) return;
  lastSensorRead = now;
  if (!sensorData.bmeOk) return;
  float t = bme.readTemperature();
  float h = bme.readHumidity();
  float p = bme.readPressure() / 100.0f;
  if (isnan(t) || isnan(h) || isnan(p) || p < 200.0f || p > 1300.0f || h < 0.0f || h > 100.0f || t < -100.0f || t > 200.0f) {
    sensorData.valid = false;
    return;
  }
  sensorData.temperature = t;
  sensorData.humidity = h;
  sensorData.pressure = p;
  sensorData.altitude = bme.readAltitude(settings.altRef);
  sensorData.valid = true;
  sensorData.lastRead = now;
  evaluateRelay();
}

void handleApiData() {

  uint32_t uptime = (uint32_t)(millis() / 1000UL);
  uint32_t heap = ESP.getFreeHeap();
  int rssi = WiFi.RSSI();
  float t = sensorData.valid ? sensorData.temperature : 0.0f;
  float h = sensorData.valid ? sensorData.humidity : 0.0f;
  float p = sensorData.valid ? sensorData.pressure : 0.0f;
  float a = sensorData.valid ? sensorData.altitude : 0.0f;
  snprintf(buf, sizeof(buf),
    "{\"t\":%.2f,\"h\":%.2f,\"p\":%.2f,\"a\":%.2f,\"r\":%d,\"u\":%lu,\"hp\":%lu,\"s\":%d,\"v\":%d}",
    t, h, p, a, relayState ? 1 : 0, (unsigned long)uptime, (unsigned long)heap, rssi,
    sensorData.valid ? 1 : 0);
  sendJSONRaw(200, buf);
}

void handleApiSettingsGet() {
  char buf[256];
  snprintf(buf, sizeof(buf),
    "{\"tmin\":%.2f,\"tmax\":%.2f,\"hmin\":%.2f,\"hmax\":%.2f,\"pmin\":%.2f,\"pmax\":%.2f,\"altref\":%.2f,\"rmode\":%u}",
    settings.tempMin, settings.tempMax, settings.humMin, settings.humMax,
    settings.presMin, settings.presMax, settings.altRef, (unsigned)settings.relayMode);
  sendJSONRaw(200, buf);
}

void handleApiSettingsPost() {
  if (!server.hasArg("plain")) { sendJSONErr("no body"); return; }
  String body = server.arg("plain");
  if (body.length() > 512) { sendJSONErr("body too large"); return; }
  float tmin, tmax, hmin, hmax, pmin, pmax, altref;
  int rmode;
  int n = sscanf(body.c_str(),
    "{\"tmin\":%f,\"tmax\":%f,\"hmin\":%f,\"hmax\":%f,\"pmin\":%f,\"pmax\":%f,\"altref\":%f,\"rmode\":%d}",
    &tmin, &tmax, &hmin, &hmax, &pmin, &pmax, &altref, &rmode);
  if (n != 8) { sendJSONErr("parse error"); return; }
  Settings ns;
  ns.tempMin = tmin; ns.tempMax = tmax;
  ns.humMin = hmin; ns.humMax = hmax;
  ns.presMin = pmin; ns.presMax = pmax;
  ns.altRef = altref; ns.relayMode = (uint8_t)rmode;
  if (!settingsValid(ns)) { sendJSONErr("invalid values"); return; }
  settings = ns;
  nvsSaveSettings();
  sensorData.altitude = sensorData.bmeOk ? bme.readAltitude(settings.altRef) : 0.0f;
  evaluateRelay();
  sendJSONOk();
}

void handleApiWifiPost() {
  if (!server.hasArg("plain")) { sendJSONErr("no body"); return; }
  String body = server.arg("plain");
  if (body.length() > 256) { sendJSONErr("body too large"); return; }
  char ssid[64] = {0};
  char pass[80] = {0};
  const char *cp = body.c_str();
  const char *sp = strstr(cp, "\"ssid\":\"");
  if (!sp) { sendJSONErr("missing ssid"); return; }
  sp += 8;
  const char *se = strchr(sp, '"');
  if (!se) { sendJSONErr("ssid parse"); return; }
  size_t sl = (size_t)(se - sp);
  if (sl > 32) { sendJSONErr("ssid too long"); return; }
  memcpy(ssid, sp, sl); ssid[sl] = 0;
  const char *pp = strstr(se + 1, "\"pass\":\"");
  if (pp) {
    pp += 8;
    const char *pe = strchr(pp, '"');
    if (!pe) { sendJSONErr("pass parse"); return; }
    size_t pl = (size_t)(pe - pp);
    if (pl > 63) { sendJSONErr("pass too long"); return; }
    memcpy(pass, pp, pl); pass[pl] = 0;
  }
  size_t slen = strlen(ssid);
  size_t plen = strlen(pass);
  if (slen < 1 || slen > 32) { sendJSONErr("ssid length"); return; }
  if (plen != 0 && (plen < 8 || plen > 63)) { sendJSONErr("pass length"); return; }
  for (size_t i = 0; i < slen; i++) {
    unsigned char c = (unsigned char)ssid[i];
    if (c < 0x20 || c == 0x7f || c == '"' || c == '\\') { sendJSONErr("ssid char"); return; }
  }
  for (size_t i = 0; i < plen; i++) {
    unsigned char c = (unsigned char)pass[i];
    if (c < 0x20 || c == 0x7f || c == '"' || c == '\\') { sendJSONErr("pass char"); return; }
  }
  strncpy(wifiConfig.ssid, ssid, 32); wifiConfig.ssid[32] = 0;
  strncpy(wifiConfig.pass, pass, 63); wifiConfig.pass[63] = 0;
  nvsSaveWifi();
  if (relayPersistPending) { nvsSaveRelayState(); relayPersistPending = false; }
  sendJSONOk();
  server.client().flush();
  delay(300);
  ESP.restart();
}

void checkBootButton() {
  if (bootResetMagic != BOOT_RESET_MAGIC) {
    bootResetMagic = BOOT_RESET_MAGIC;
    bootResetCount = 0;
  }
  int state = digitalRead(BOOT_BUTTON_PIN);
  unsigned long now = millis();
  if (state == LOW) {
    if (!bootButtonPressed) {
      bootButtonPressed = true;
      bootButtonPressStart = now;
    } else if ((unsigned long)(now - bootButtonPressStart) >= FACTORY_RESET_HOLD_MS) {
      if (bootResetCount >= BOOT_RESET_MAX_PER_POWERON) {
        bootButtonPressed = false;
        return;
      }
      bootResetCount++;
      if (relayPersistPending) { nvsSaveRelayState(); relayPersistPending = false; }
      nvsFactoryReset();
      delay(100);
      ESP.restart();
    }
  } else {
    bootButtonPressed = false;
  }
}

void setup() {
  pinMode(RELAY_PIN, OUTPUT);
  digitalWrite(RELAY_PIN, LOW);
  pinMode(BOOT_BUTTON_PIN, INPUT_PULLUP);

  nvsLoadWifi();
  nvsLoadSettings();
  nvsLoadRelayState();

  Wire.begin(I2C_SDA, I2C_SCL);
  Wire.setClock(100000);
  sensorData.bmeOk = false;
  if (bme.begin(0x76, &Wire) || bme.begin(0x77, &Wire)) {
    bme.setSampling(Adafruit_BME280::MODE_NORMAL,
                    Adafruit_BME280::SAMPLING_X2,
                    Adafruit_BME280::SAMPLING_X16,
                    Adafruit_BME280::SAMPLING_X1,
                    Adafruit_BME280::FILTER_X16,
                    Adafruit_BME280::STANDBY_MS_500);
    sensorData.bmeOk = true;
  }

  WiFi.persistent(false);
  WiFi.mode(WIFI_AP);
  WiFi.softAP(wifiConfig.ssid, strlen(wifiConfig.pass) > 0 ? wifiConfig.pass : NULL);
  delay(100);

  IPAddress apIP = WiFi.softAPIP();
  dnsServer.setErrorReplyCode(DNSReplyCode::NoError);
  dnsServer.start(53, "*", apIP);

  server.on("/", HTTP_GET, handleRoot);
  server.on("/settings", HTTP_GET, handleSettingsPage);
  server.on("/wifi", HTTP_GET, handleWifiPage);
  server.on("/api/data", HTTP_GET, handleApiData);
  server.on("/api/settings", HTTP_GET, handleApiSettingsGet);
  server.on("/api/settings", HTTP_POST, handleApiSettingsPost);
  server.on("/api/wifi", HTTP_GET, handleApiWifiGet);
  server.on("/api/wifi", HTTP_POST, handleApiWifiPost);
  server.on("/api/relay", HTTP_GET, handleApiRelay);
  server.on("/generate_204", HTTP_GET, handleCaptivePortal);
  server.on("/gen_204", HTTP_GET, handleCaptivePortal);
  server.on("/hotspot-detect.html", HTTP_GET, handleCaptivePortal);
  server.on("/library/test/success.html", HTTP_GET, handleCaptivePortal);
  server.on("/connecttest.txt", HTTP_GET, handleCaptivePortal);
  server.on("/ncsi.txt", HTTP_GET, handleCaptivePortal);
  server.on("/fwlink", HTTP_GET, handleCaptivePortal);
  server.on("/redirect", HTTP_GET, handleCaptivePortal);
  server.on("/canonical.html", HTTP_GET, handleCaptivePortal);
  server.onNotFound(handleNotFound);
  server.begin();

  digitalWrite(RELAY_PIN, relayState ? HIGH : LOW);

  esp_task_wdt_init(WDT_TIMEOUT_S, true);
  esp_task_wdt_add(NULL);

  lastSensorRead = millis() - SENSOR_READ_INTERVAL_MS;
}

void loop() {
  dnsServer.processNextRequest();
  server.handleClient();
  checkBootButton();
  readSensor();
  if (relayPersistPending && (long)(millis() - relayPersistDeadline) >= 0) {
    nvsSaveRelayState();
    relayPersistPending = false;
  }
  esp_task_wdt_reset();
  delay(2);
}
